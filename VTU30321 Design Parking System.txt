Design Parking System
https://leetcode.com/problems/design-parking-system/submissions/2138102715/
Program:-
class ParkingSystem {
    private int big;
    private int medium;
    private int small;

    // Constructor
    public ParkingSystem(int big, int medium, int small) {
        this.big = big;
        this.medium = medium;
        this.small = small;
    }

    // Add a car to the parking lot
    public boolean addCar(int carType) {
        if (carType == 1) {
            if (big > 0) {
                big--;
                return true;
            }
        } 
        else if (carType == 2) {
            if (medium > 0) {
                medium--;
                return true;
            }
        } 
        else if (carType == 3) {
            if (small > 0) {
                small--;
                return true;
            }
        }

        return false;
    }
}
